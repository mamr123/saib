# 1. Set the CP4BA namespace
export NS=bawuat

echo "Using namespace: $NS"
oc project "$NS"


# 3. Check existing ZenExtensions before changing anything
oc get zenextension -n "$NS"



# 4. Back up an existing same-name extension, if one exists
if oc get zenextension samesite-none-extension -n "$NS" >/dev/null 2>&1; then
  echo "Existing samesite-none-extension found. Creating backup."
  oc get zenextension samesite-none-extension -n "$NS" -o yaml \
    > samesite-none-extension.backup.yaml
else
  echo "No existing samesite-none-extension found."
fi



# 5. Create the YAML file
cat > samesite-none-extension.yaml <<'EOF'
apiVersion: zen.cpd.ibm.com/v1
kind: ZenExtension
metadata:
  name: samesite-none-extension
  namespace: bawuat
spec:
  samesite-none.conf: |
    header_filter_by_lua_block {
        local headers = ngx.header["Set-Cookie"]

        if headers then
            local function modify_cookie(cookie)
                if cookie:match("SameSite=Strict") or cookie:match("SameSite=Lax") then
                    local modified = cookie:gsub("SameSite=Strict", "SameSite=None")
                                          :gsub("SameSite=Lax", "SameSite=None")

                    if not modified:match("Secure") then
                        modified = modified .. "; Secure"
                    end

                    return modified
                end
                return cookie
            end

            if type(headers) == "table" then
                for i, cookie in ipairs(headers) do
                    headers[i] = modify_cookie(cookie)
                end
                ngx.header["Set-Cookie"] = headers
            else
                ngx.header["Set-Cookie"] = modify_cookie(headers)
            end
        end
    }
  extensions: |
    [
      {
          "extension_point_id": "zen_front_door",
          "extension_name": "samesite-none-extension",
          "details": {
            "upstream_conf": "samesite-none.conf"
          }
      }
    ]
EOF


oc delete pod -n bawuat -l component=ibm-nginx
