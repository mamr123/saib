oc get zenextension samesite-none-extension -n "$NS" \
  -o yaml > samesite-none-extension.cluster-backup.yaml






cat > samesite-frame-ancestors-extension.yaml <<'EOF'
apiVersion: zen.cpd.ibm.com/v1
kind: ZenExtension
metadata:
  name: samesite-none-extension
  namespace: bawuat
spec:
  samesite-none.conf: |
    header_filter_by_lua_block {
        local headers = ngx.header["Set-Cookie"]

        if headers then
            local function modify_cookie(cookie)
                if cookie:match("SameSite=Strict") or cookie:match("SameSite=Lax") then
                    local modified = cookie:gsub("SameSite=Strict", "SameSite=None")
                                          :gsub("SameSite=Lax", "SameSite=None")

                    if not modified:match("Secure") then
                        modified = modified .. "; Secure"
                    end

                    return modified
                end
                return cookie
            end

            if type(headers) == "table" then
                for i, cookie in ipairs(headers) do
                    headers[i] = modify_cookie(cookie)
                end
                ngx.header["Set-Cookie"] = headers
            else
                ngx.header["Set-Cookie"] = modify_cookie(headers)
            end
        end

        local uri = ngx.var.uri or ""

        if uri:match("^/baw%-bawins1/ProcessPortal") then
            local csp = ngx.header["Content-Security-Policy"]

            local allowed_frame_ancestors = {
                "http://crmcauat365.saib.com.sa",
                "https://crmcauat365.saib.com.sa"
            }

            local function add_frame_ancestors(policy)
                if type(policy) ~= "string" then
                    return policy
                end

                if not policy:find("frame%-ancestors") then
                    policy = policy ..
                        "; frame-ancestors " ..
                        table.concat(allowed_frame_ancestors, " ")
                    return policy
                end

                for _, origin in ipairs(allowed_frame_ancestors) do
                    if not policy:find(origin, 1, true) then
                        policy = policy:gsub(
                            "frame%-ancestors%s+",
                            "frame-ancestors " .. origin .. " ",
                            1
                        )
                    end
                end

                return policy
            end

            if type(csp) == "table" then
                for i, policy in ipairs(csp) do
                    csp[i] = add_frame_ancestors(policy)
                end
                ngx.header["Content-Security-Policy"] = csp
            elseif csp then
                ngx.header["Content-Security-Policy"] =
                    add_frame_ancestors(csp)
            end
        end
    }
  extensions: |
    [
      {
          "extension_point_id": "zen_front_door",
          "extension_name": "samesite-none-extension",
          "details": {
            "upstream_conf": "samesite-none.conf"
          }
      }
    ]
EOF
