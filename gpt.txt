# Get the catalog pod name
CAT_POD=$(oc get po -n bawdev -l olm.catalogSource=ibm-iam-operator-catalog-4-16-0 -o jsonpath='{.items[0].metadata.name}')

# Copy the index DB out
oc cp bawdev/$CAT_POD:/database/index.db ./index.db

# Query it locally
sqlite3 ./index.db "SELECT name FROM package WHERE name LIKE '%iam%';"
sqlite3 ./index.db "SELECT name, head_operatorbundle_name FROM channel WHERE package_name='ibm-iam-operator';"
sqlite3 ./index.db "SELECT name, version, replaces FROM operatorbundle WHERE package_name='ibm-iam-operator';"
