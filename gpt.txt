  -l component=ibm-nginx \
  -o jsonpath='{.items[*].metadata.name}'); do

  CONTAINER=$(oc get pod "$POD" -n "$NS" \
    -o jsonpath='{.spec.containers[0].name}')

  echo "===== $POD / $CONTAINER ====="

  oc exec -n "$NS" "$POD" -c "$CONTAINER" -- sh -c \
    'nginx -T 2>/dev/null |
     grep -n -i -A4 -B4 "frame-ancestors\|content-security-policy"'

python3 - <<'PY'
import requests
import urllib3
from urllib.parse import urlsplit

urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)

url = "https://cpd-bawuat.apps.cp4bauatsaibocp.saibnet2.saib.com/baw-bawins1/ProcessPortal/"

def clean_url(value):
    if not value:
        return ""
    parsed = urlsplit(value)
    return f"{parsed.scheme}://{parsed.netloc}{parsed.path}"

session = requests.Session()
response = session.get(
    url,
    verify=False,
    timeout=30,
    allow_redirects=True
)

chain = list(response.history) + [response]

for index, item in enumerate(chain):
    print(f"\n===== Response {index} =====")
    print("URL:", clean_url(item.url))
    print("Status:", item.status_code)
    print("Content-Security-Policy:", item.headers.get("Content-Security-Policy"))
    print("X-Frame-Options:", item.headers.get("X-Frame-Options"))
    print("Location:", clean_url(item.headers.get("Location")))
